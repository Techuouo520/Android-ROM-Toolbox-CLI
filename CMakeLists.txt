cmake_minimum_required(VERSION 3.24)
if(CMAKE_HOST_WIN32 AND NOT CMAKE_TOOLCHAIN_FILE AND NOT ARTB_TARGET)
    if(NOT DEFINED CMAKE_C_COMPILER AND NOT DEFINED ENV{CC})
        set(CMAKE_C_COMPILER clang)
    endif()
    if(NOT DEFINED CMAKE_CXX_COMPILER AND NOT DEFINED ENV{CXX})
        set(CMAKE_CXX_COMPILER clang++)
    endif()
endif()
if(CMAKE_HOST_SYSTEM_NAME STREQUAL "Darwin" AND NOT DEFINED CMAKE_OSX_DEPLOYMENT_TARGET)
    set(CMAKE_OSX_DEPLOYMENT_TARGET "15.0" CACHE STRING "Minimum macOS version")
endif()
project(artb_native LANGUAGES C CXX)
if(NOT CMAKE_CONFIGURATION_TYPES AND NOT CMAKE_BUILD_TYPE)
    set(CMAKE_BUILD_TYPE Release CACHE STRING "Build type" FORCE)
endif()
set(CMAKE_C_STANDARD 17)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
if(WIN32)
    enable_language(RC)
    set(ARTB_UTF8_MANIFEST_RC "${CMAKE_CURRENT_LIST_DIR}/native/windows/utf8.rc")
endif()
option(ARTB_BUILD_BUNDLE "Build the Go CLI with embedded native tools" ON)
option(ARTB_BUILD_ANDROID_TOOLS "Build Android sparse and logical partition tools" ${ARTB_BUILD_BUNDLE})
option(ARTB_BUILD_EROFS_TOOLS "Build the EROFS extraction tool" ${ARTB_BUILD_BUNDLE})
option(ARTB_BUILD_EXT4_TOOLS "Build the EXT4 extraction tool" ${ARTB_BUILD_BUNDLE})
option(ARTB_BUILD_BOOT_TOOLS "Build the boot image tool" ${ARTB_BUILD_BUNDLE})
option(ARTB_BUILD_F2FS_TOOLS "Build the F2FS image tools (needs the Android, EROFS and EXT4 tools)" ${ARTB_BUILD_EROFS_TOOLS})
set(ARTB_STATIC_NATIVE_DEFAULT ON)
if(APPLE)
    set(ARTB_STATIC_NATIVE_DEFAULT OFF)
endif()
option(ARTB_STATIC_NATIVE "Link every native tool fully static so it needs nothing beyond the OS kernel and system DLLs" ${ARTB_STATIC_NATIVE_DEFAULT})
set(ARTB_NATIVE_ONLY_DEFAULT ${ARTB_BUILD_BUNDLE})
if(ARTB_TARGET)
    set(ARTB_NATIVE_ONLY_DEFAULT OFF)
endif()
option(ARTB_NATIVE_ONLY "Require the bundle operating system and architecture to match the build host" ${ARTB_NATIVE_ONLY_DEFAULT})
if(ARTB_NATIVE_ONLY AND WIN32 AND NOT CMAKE_GENERATOR STREQUAL "Ninja")
    message(FATAL_ERROR "Windows native builds require Ninja; configure with cmake -G Ninja ..")
endif()
if(ARTB_NATIVE_ONLY AND WIN32 AND (NOT MINGW OR NOT CMAKE_C_COMPILER_ID STREQUAL "Clang"))
    message(FATAL_ERROR "Windows host builds require MSYS2 UCRT64 Clang ahead of other compilers on PATH")
endif()
if(ARTB_STATIC_NATIVE)
    if(APPLE)
        message(FATAL_ERROR "macOS has no static libSystem; configure with -DARTB_STATIC_NATIVE=OFF")
    endif()
    # Applies to every tool added below, so future tools are static by default; Release builds also drop symbol tables.
    set(CMAKE_FIND_LIBRARY_SUFFIXES ".a")
    set(ZLIB_USE_STATIC_LIBS TRUE)
    add_link_options(-static -static-libgcc -static-libstdc++ "$<$<CONFIG:Release,MinSizeRel>:-s>")
endif()
add_subdirectory(native/common common)
if(ARTB_BUILD_ANDROID_TOOLS OR ARTB_BUILD_EXT4_TOOLS)
    find_package(ZLIB)
    if(NOT ZLIB_FOUND)
        if(NOT ARTB_BUILD_EROFS_TOOLS)
            message(FATAL_ERROR "No zlib for this target; enable ARTB_BUILD_EROFS_TOOLS to build the vendored one")
        endif()
        # Toolchains without zlib, such as llvm-mingw for Windows ARM64, link the vendored Chromium zlib that the EROFS tools build.
        add_library(ZLIB::ZLIB INTERFACE IMPORTED)
        set_target_properties(ZLIB::ZLIB PROPERTIES INTERFACE_LINK_LIBRARIES z_static)
        set(ZLIB_INCLUDE_DIRS "${CMAKE_CURRENT_LIST_DIR}/upstream/erofs-tools/src/lib/zlib")
    endif()
endif()
if(ARTB_BUILD_ANDROID_TOOLS)
    add_subdirectory(native/android-tools android-tools)
endif()
if(ARTB_BUILD_EROFS_TOOLS)
    add_subdirectory(native/erofs-tools erofs-tools)
endif()
if(ARTB_BUILD_EXT4_TOOLS)
    add_subdirectory(native/ext4 ext4)
endif()
if(ARTB_BUILD_F2FS_TOOLS)
    add_subdirectory(native/f2fs-tools f2fs-tools)
endif()
if(ARTB_BUILD_BOOT_TOOLS)
    add_subdirectory(native/magiskboot magiskboot)
    add_subdirectory(native/dtc dtc)
endif()
if(ARTB_BUILD_BUNDLE)
    if(NOT ARTB_BUILD_ANDROID_TOOLS OR NOT ARTB_BUILD_EROFS_TOOLS OR NOT ARTB_BUILD_EXT4_TOOLS OR NOT ARTB_BUILD_F2FS_TOOLS OR NOT ARTB_BUILD_BOOT_TOOLS)
        message(FATAL_ERROR "ARTB_BUILD_BUNDLE requires ARTB_BUILD_ANDROID_TOOLS=ON, ARTB_BUILD_EROFS_TOOLS=ON, ARTB_BUILD_EXT4_TOOLS=ON, ARTB_BUILD_F2FS_TOOLS=ON and ARTB_BUILD_BOOT_TOOLS=ON")
    endif()
    find_program(ARTB_GO_EXECUTABLE go REQUIRED)
    if(CMAKE_SYSTEM_NAME STREQUAL "Windows")
        set(ARTB_GOOS windows)
    elseif(CMAKE_SYSTEM_NAME STREQUAL "Linux")
        set(ARTB_GOOS linux)
    elseif(CMAKE_SYSTEM_NAME STREQUAL "Darwin")
        set(ARTB_GOOS darwin)
    elseif(CMAKE_SYSTEM_NAME STREQUAL "Android")
        set(ARTB_GOOS android)
    else()
        message(FATAL_ERROR "Unsupported bundle operating system: ${CMAKE_SYSTEM_NAME}")
    endif()
    if(CMAKE_SYSTEM_PROCESSOR MATCHES "^(AMD64|amd64|x86_64)$")
        set(ARTB_GOARCH amd64)
    elseif(CMAKE_SYSTEM_PROCESSOR MATCHES "^(ARM64|arm64|aarch64)$")
        set(ARTB_GOARCH arm64)
    else()
        message(FATAL_ERROR "Unsupported bundle architecture: ${CMAKE_SYSTEM_PROCESSOR}")
    endif()
    if(ARTB_NATIVE_ONLY)
        if(CMAKE_CROSSCOMPILING OR NOT CMAKE_SYSTEM_NAME STREQUAL CMAKE_HOST_SYSTEM_NAME)
            message(FATAL_ERROR "The host preset only builds for this machine; use a target preset for cross-compilation")
        endif()
        if(CMAKE_HOST_SYSTEM_PROCESSOR MATCHES "^(AMD64|amd64|x86_64)$")
            set(ARTB_HOST_ARCH amd64)
        elseif(CMAKE_HOST_SYSTEM_PROCESSOR MATCHES "^(ARM64|arm64|aarch64)$")
            set(ARTB_HOST_ARCH arm64)
        else()
            message(FATAL_ERROR "Unsupported host architecture: ${CMAKE_HOST_SYSTEM_PROCESSOR}")
        endif()
        if(NOT ARTB_GOARCH STREQUAL ARTB_HOST_ARCH)
            message(FATAL_ERROR "The host preset selected ${ARTB_GOARCH}, but this machine is ${ARTB_HOST_ARCH}")
        endif()
    endif()
    set(ARTB_TARGET "" CACHE STRING "Expected <goos>_<goarch> of the bundle; the presets set it so a preset run with the wrong toolchain or host fails at configure time")
    if(ARTB_TARGET AND NOT ARTB_TARGET STREQUAL "${ARTB_GOOS}_${ARTB_GOARCH}")
        message(FATAL_ERROR "Preset target ${ARTB_TARGET} does not match this toolchain, which builds ${ARTB_GOOS}_${ARTB_GOARCH}")
    endif()
    set(ARTB_GO_ROOT "${CMAKE_CURRENT_LIST_DIR}")
    set(ARTB_PAYLOAD_ROOT "${ARTB_GO_ROOT}/upstream/payload-dumper-go")
    set(ARTB_LZMA_SDK_ROOT "${ARTB_GO_ROOT}/upstream/lzma-sdk")
    if(NOT EXISTS "${ARTB_PAYLOAD_ROOT}/payload/extract.go" OR NOT EXISTS "${ARTB_LZMA_SDK_ROOT}/C/XzDec.c" OR NOT EXISTS "${ARTB_LZMA_SDK_ROOT}/Asm/x86/LzmaDecOpt.asm")
        message(FATAL_ERROR "Payload or LZMA SDK sources missing under ${ARTB_GO_ROOT}/upstream")
    endif()
    # LzmaDecOpt.asm is MASM syntax: ml64 emits COFF on Windows, the vendored UASM emits ELF with the SysV ABI on Linux; arm64 uses the SDK's GNU-syntax LzmaDecOpt.S.
    set(ARTB_GO_TAGS nativeembed)
    set(ARTB_LZMA_ASM_OUTPUTS)
    set(ARTB_MASM_DEPENDS)
    if(ARTB_GOARCH STREQUAL "amd64" AND ARTB_GOOS STREQUAL "windows")
        file(GLOB ARTB_ML64_HINTS "$ENV{ProgramFiles}/Microsoft Visual Studio/*/*/VC/Tools/MSVC/*/bin/Hostx64/x64")
        find_program(ARTB_MASM_EXECUTABLE NAMES ml64 HINTS ${ARTB_ML64_HINTS})
        set(ARTB_MASM_FLAGS -nologo -c)
    elseif(ARTB_GOARCH STREQUAL "amd64" AND ARTB_GOOS STREQUAL "linux" AND NOT CMAKE_CROSSCOMPILING)
        # Built from upstream/uasm so Linux builds never depend on an assembler installed on the host.
        add_subdirectory(native/uasm uasm)
        set(ARTB_MASM_EXECUTABLE "$<TARGET_FILE:artb_uasm>")
        set(ARTB_MASM_DEPENDS artb_uasm)
        set(ARTB_MASM_FLAGS -nologo -c -elf64 -DABI_LINUX)
    endif()
    if(ARTB_MASM_EXECUTABLE)
        set(ARTB_LZMA_SYSO "${ARTB_GO_ROOT}/xz/lzmadecopt_${ARTB_GOOS}_${ARTB_GOARCH}.syso")
        set(ARTB_LZMA_STACK_COMMAND)
        if(ARTB_GOOS STREQUAL "linux")
            find_program(ARTB_OBJCOPY_EXECUTABLE NAMES objcopy llvm-objcopy REQUIRED)
            set(ARTB_GNU_STACK_NOTE "${CMAKE_CURRENT_BINARY_DIR}/artb-gnu-stack-note")
            file(WRITE "${ARTB_GNU_STACK_NOTE}" "")
            set(ARTB_LZMA_STACK_COMMAND COMMAND "${ARTB_OBJCOPY_EXECUTABLE}" --add-section ".note.GNU-stack=${ARTB_GNU_STACK_NOTE}" "${ARTB_LZMA_SYSO}")
        endif()
        add_custom_command(OUTPUT "${ARTB_LZMA_SYSO}" COMMAND "${ARTB_MASM_EXECUTABLE}" ${ARTB_MASM_FLAGS} "-Fo${ARTB_LZMA_SYSO}" LzmaDecOpt.asm ${ARTB_LZMA_STACK_COMMAND} WORKING_DIRECTORY "${ARTB_LZMA_SDK_ROOT}/Asm/x86" DEPENDS "${ARTB_LZMA_SDK_ROOT}/Asm/x86/LzmaDecOpt.asm" "${ARTB_LZMA_SDK_ROOT}/Asm/x86/7zAsm.asm" ${ARTB_MASM_DEPENDS} VERBATIM)
        list(APPEND ARTB_GO_TAGS artb_lzmaasm)
        list(APPEND ARTB_LZMA_ASM_OUTPUTS "${ARTB_LZMA_SYSO}")
    elseif(ARTB_GOARCH STREQUAL "arm64" AND ARTB_GOOS MATCHES "^(linux|android|darwin)$")
        # cgo assembles xz/lzmadecopt_arm64.S with the target C compiler, so no separate assembler is needed.
        list(APPEND ARTB_GO_TAGS artb_lzmaasm)
    else()
        message(STATUS "LzmaDecOpt assembler unavailable for ${ARTB_GOOS}/${ARTB_GOARCH}; using the C LZMA decoder")
    endif()
    set(ARTB_GO_LDFLAGS "-s -w")
    set(ARTB_GO_CC "${CMAKE_C_COMPILER}")
    set(ARTB_GO_CXX "${CMAKE_CXX_COMPILER}")
    if(ARTB_GOOS STREQUAL "darwin")
        set(ARTB_GO_LDFLAGS "-s -w -macos=${CMAKE_OSX_DEPLOYMENT_TARGET}")
        set(ARTB_GO_DARWIN_ENV "MACOSX_DEPLOYMENT_TARGET=${CMAKE_OSX_DEPLOYMENT_TARGET}" "CGO_CFLAGS=-mmacosx-version-min=${CMAKE_OSX_DEPLOYMENT_TARGET}" "CGO_CXXFLAGS=-mmacosx-version-min=${CMAKE_OSX_DEPLOYMENT_TARGET}" "CGO_LDFLAGS=-mmacosx-version-min=${CMAKE_OSX_DEPLOYMENT_TARGET}")
    endif()
    if(ANDROID)
        # Go calls the NDK clang itself, so it needs the triple CMake passes implicitly.
        set(ARTB_GO_CC "\"${CMAKE_C_COMPILER}\" --target=${ANDROID_LLVM_TRIPLE}")
        set(ARTB_GO_CXX "\"${CMAKE_CXX_COMPILER}\" --target=${ANDROID_LLVM_TRIPLE}")
    endif()
    # Android's cgo runtime dlopens libc.so at startup to find its TLS slot, so there the Go program links only the platform's libc, libdl and liblog while the embedded tools stay static.
    if(ARTB_STATIC_NATIVE AND ARTB_GOOS STREQUAL "linux")
        list(APPEND ARTB_GO_TAGS netgo osusergo)
        set(ARTB_GO_LDFLAGS "-s -w -linkmode external -extldflags=-static")
    endif()
    # gozstd bundles a prebuilt libzstd for every target here except windows/arm64, which links the vendored zstd 1.6.0; its API matches gozstd's 1.5.7 headers apart from comments and one added function.
    set(ARTB_GO_ZSTD_ENV)
    set(ARTB_GO_ZSTD_DEPENDS)
    if(ARTB_GOOS STREQUAL "windows" AND ARTB_GOARCH STREQUAL "arm64")
        set(ARTB_GO_ZSTD_ENV "CGO_LDFLAGS=$<TARGET_FILE:libzstd_static>")
        set(ARTB_GO_ZSTD_DEPENDS libzstd_static)
    endif()
    # artb version and the tool cache folder use the git tag, else the short commit, with -dirty when configured over uncommitted changes; -DARTB_VERSION overrides it.
    set(ARTB_VERSION_STAMP "${ARTB_VERSION}")
    find_package(Git QUIET)
    if(NOT ARTB_VERSION_STAMP AND GIT_FOUND)
        execute_process(COMMAND "${GIT_EXECUTABLE}" describe --tags --always --dirty WORKING_DIRECTORY "${ARTB_GO_ROOT}" OUTPUT_VARIABLE ARTB_VERSION_STAMP OUTPUT_STRIP_TRAILING_WHITESPACE ERROR_QUIET)
        execute_process(COMMAND "${GIT_EXECUTABLE}" rev-parse --path-format=absolute --git-path HEAD WORKING_DIRECTORY "${ARTB_GO_ROOT}" OUTPUT_VARIABLE ARTB_GIT_HEAD OUTPUT_STRIP_TRAILING_WHITESPACE ERROR_QUIET)
        execute_process(COMMAND "${GIT_EXECUTABLE}" symbolic-ref -q HEAD WORKING_DIRECTORY "${ARTB_GO_ROOT}" OUTPUT_VARIABLE ARTB_GIT_BRANCH OUTPUT_STRIP_TRAILING_WHITESPACE ERROR_QUIET)
        if(ARTB_GIT_BRANCH)
            execute_process(COMMAND "${GIT_EXECUTABLE}" rev-parse --path-format=absolute --git-path "${ARTB_GIT_BRANCH}" WORKING_DIRECTORY "${ARTB_GO_ROOT}" OUTPUT_VARIABLE ARTB_GIT_REF OUTPUT_STRIP_TRAILING_WHITESPACE ERROR_QUIET)
        endif()
        # A commit or checkout rewrites HEAD or the branch ref, so the next build reconfigures and restamps.
        foreach(ARTB_GIT_FILE IN ITEMS "${ARTB_GIT_HEAD}" "${ARTB_GIT_REF}")
            if(ARTB_GIT_FILE AND EXISTS "${ARTB_GIT_FILE}")
                set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${ARTB_GIT_FILE}")
            endif()
        endforeach()
    endif()
    if(NOT ARTB_VERSION_STAMP)
        set(ARTB_VERSION_STAMP dev)
    endif()
    # The version names a cache folder, so anything outside a plain file-name character set becomes a dash.
    string(REGEX REPLACE "[^0-9A-Za-z._+-]" "-" ARTB_VERSION_STAMP "${ARTB_VERSION_STAMP}")
    message(STATUS "artb version ${ARTB_VERSION_STAMP}")
    string(APPEND ARTB_GO_LDFLAGS " -X artb/internal/buildinfo.Version=${ARTB_VERSION_STAMP}")
    list(JOIN ARTB_GO_TAGS "," ARTB_GO_TAGS_ARG)
    set(ARTB_ARCHCHECK_FLAGS "-goos=${ARTB_GOOS}" "-goarch=${ARTB_GOARCH}")
    if(ARTB_STATIC_NATIVE)
        list(APPEND ARTB_ARCHCHECK_FLAGS -static)
    endif()
    # Each target stages its tools in its own folder and internal/nativebin embeds only the folder matching GOOS and GOARCH, so builds for different targets never share files.
    set(ARTB_EMBED_DIR "${ARTB_GO_ROOT}/internal/nativebin/files/${ARTB_GOOS}_${ARTB_GOARCH}")
    set(ARTB_OUTPUT_DIR "${CMAKE_CURRENT_LIST_DIR}/dist" CACHE PATH "Bundle output directory")
    set(ARTB_OUTPUT "${ARTB_OUTPUT_DIR}/artb-${ARTB_GOOS}-${ARTB_GOARCH}${CMAKE_EXECUTABLE_SUFFIX}")
    file(GLOB_RECURSE ARTB_GO_SOURCES CONFIGURE_DEPENDS "${ARTB_GO_ROOT}/cmd/*.go" "${ARTB_GO_ROOT}/internal/*.go" "${ARTB_GO_ROOT}/upstream/payload-dumper-go/*.go" "${ARTB_GO_ROOT}/xz/*.go")
    file(GLOB ARTB_XZ_SOURCES CONFIGURE_DEPENDS "${ARTB_GO_ROOT}/xz/*.c" "${ARTB_GO_ROOT}/xz/*.h" "${ARTB_GO_ROOT}/xz/*.S" "${ARTB_LZMA_SDK_ROOT}/Asm/arm64/*.S" "${ARTB_GO_ROOT}/xz/go.mod" "${ARTB_LZMA_SDK_ROOT}/C/*.c" "${ARTB_LZMA_SDK_ROOT}/C/*.h")
    add_custom_command(OUTPUT "${ARTB_OUTPUT}" COMMAND "${CMAKE_COMMAND}" -E remove_directory "${ARTB_EMBED_DIR}" COMMAND "${CMAKE_COMMAND}" -E make_directory "${ARTB_EMBED_DIR}" COMMAND "${CMAKE_COMMAND}" -E copy_if_different "$<TARGET_FILE:simg2img>" "$<TARGET_FILE:img2simg>" "$<TARGET_FILE:lpunpack>" "$<TARGET_FILE:lpmake>" "$<TARGET_FILE:extract.erofs>" "$<TARGET_FILE:extract.ext4>" "$<TARGET_FILE:magiskboot>" "$<TARGET_FILE:artb_mkfs_erofs>" "$<TARGET_FILE:mke2fs>" "$<TARGET_FILE:e2fsdroid>" "$<TARGET_FILE:extract.f2fs>" "$<TARGET_FILE:artb_mkfs_f2fs>" "$<TARGET_FILE:artb_sload_f2fs>" "$<TARGET_FILE:dtc>" "${ARTB_EMBED_DIR}" COMMAND "${CMAKE_COMMAND}" -E env --unset=GOOS --unset=GOARCH CGO_ENABLED=0 "${ARTB_GO_EXECUTABLE}" run ./internal/archcheck ${ARTB_ARCHCHECK_FLAGS} "${ARTB_EMBED_DIR}" COMMAND "${CMAKE_COMMAND}" -E make_directory "${ARTB_OUTPUT_DIR}" COMMAND "${CMAKE_COMMAND}" -E env "CGO_ENABLED=1" "GOOS=${ARTB_GOOS}" "GOARCH=${ARTB_GOARCH}" "CC=${ARTB_GO_CC}" "CXX=${ARTB_GO_CXX}" ${ARTB_GO_DARWIN_ENV} ${ARTB_GO_ZSTD_ENV} "${ARTB_GO_EXECUTABLE}" build "-tags=${ARTB_GO_TAGS_ARG}" -trimpath "-ldflags=${ARTB_GO_LDFLAGS}" -o "${ARTB_OUTPUT}" ./cmd/artb WORKING_DIRECTORY "${ARTB_GO_ROOT}" DEPENDS simg2img img2simg lpunpack lpmake extract.erofs extract.ext4 magiskboot artb_mkfs_erofs mke2fs e2fsdroid extract.f2fs artb_mkfs_f2fs artb_sload_f2fs dtc ${ARTB_GO_ZSTD_DEPENDS} ${ARTB_GO_SOURCES} ${ARTB_XZ_SOURCES} ${ARTB_LZMA_ASM_OUTPUTS} "${ARTB_GO_ROOT}/go.mod" "${ARTB_GO_ROOT}/go.sum" VERBATIM)
    add_custom_target(artb ALL DEPENDS "${ARTB_OUTPUT}")
endif()
