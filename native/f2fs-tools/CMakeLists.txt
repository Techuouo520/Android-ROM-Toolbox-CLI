set(ARTB_F2FS_ROOT "${CMAKE_CURRENT_LIST_DIR}/../../upstream/f2fs-tools")
set(ARTB_F2FS_COMPAT_DIR "${CMAKE_CURRENT_LIST_DIR}/compat")
if(NOT TARGET artb_sparse OR NOT TARGET lz4_static OR NOT TARGET cutils_static OR NOT TARGET artb_ext2_uuid)
    message(FATAL_ERROR "F2FS tools need ARTB_BUILD_ANDROID_TOOLS, ARTB_BUILD_EROFS_TOOLS and ARTB_BUILD_EXT4_TOOLS")
endif()
set(ARTB_F2FS_LIB_SOURCES lib/libf2fs.c lib/libf2fs_io.c lib/libf2fs_zoned.c lib/nls_utf8.c)
list(TRANSFORM ARTB_F2FS_LIB_SOURCES PREPEND "${ARTB_F2FS_ROOT}/")
add_library(artb_f2fs STATIC ${ARTB_F2FS_LIB_SOURCES})
target_include_directories(artb_f2fs PUBLIC "${ARTB_F2FS_COMPAT_DIR}" "${ARTB_F2FS_ROOT}/include" "${ARTB_F2FS_ROOT}/fsck" "${ARTB_F2FS_ROOT}/mkfs")
# WITH_ANDROID matches the make_f2fs and sload_f2fs that build device images; ARTB_F2FS marks the host-independence patches.
# HAVE_LIBSELINUX adds fields ahead of chksum_seed in the global config, so the lib and every tool must agree on it or inode checksums use a stale seed.
target_compile_definitions(artb_f2fs PUBLIC WITH_ANDROID ARTB_F2FS _FILE_OFFSET_BITS=64 F2FS_MAJOR_VERSION=1 F2FS_MINOR_VERSION=16 "F2FS_TOOLS_VERSION=\"1.16.0\"" "F2FS_TOOLS_DATE=\"2023-04-11\"" HAVE_LIBLZ4=1 HAVE_LIBUUID=1 HAVE_UUID_UUID_H=1 HAVE_SPARSE_SPARSE_H=1 HAVE_LIBSELINUX=1 HAVE_SELINUX_ANDROID_H=1)
target_compile_options(artb_f2fs PUBLIC -Wno-macro-redefined -Wno-missing-field-initializers -Wno-pointer-arith -Wno-sign-compare -Wno-unused-parameter -Wno-unused-function -Wno-unused-variable -Wno-unused-but-set-variable)
target_link_libraries(artb_f2fs PUBLIC artb_sparse artb_ext2_uuid lz4_static z_static artb_selabel)
if(ANDROID)
    # Host mode keeps the device-only libselinux and fs_config lookups out; files come from the tree artb passes in.
    target_compile_options(artb_f2fs PUBLIC -U__ANDROID__)
endif()
if(WIN32)
    # e2fsprogs' mingw sys/stat.h, reached through the uuid include path, only chains to the real header when told it exists.
    target_compile_definitions(artb_f2fs PUBLIC HAVE_SYS_STAT_H=1)
    target_compile_options(artb_f2fs PUBLIC "-include${ARTB_F2FS_COMPAT_DIR}/artb_f2fs_windows_compat.h")
    # Compiled into each tool so the binary-mode constructor is linked even where nothing calls scandir.
    target_sources(artb_f2fs INTERFACE "${ARTB_F2FS_COMPAT_DIR}/artb_f2fs_windows_compat.c")
    target_link_libraries(artb_f2fs PUBLIC artb_winfs ws2_32)
endif()
add_executable(artb_mkfs_f2fs "${ARTB_F2FS_ROOT}/mkfs/f2fs_format.c" "${ARTB_F2FS_ROOT}/mkfs/f2fs_format_main.c" "${ARTB_F2FS_ROOT}/mkfs/f2fs_format_utils.c")
target_link_libraries(artb_mkfs_f2fs PRIVATE artb_f2fs)
set(ARTB_FSCK_F2FS_SOURCES dict.c dir.c dump.c main.c mkquota.c mount.c node.c quotaio.c quotaio_tree.c quotaio_v2.c segment.c xattr.c compress.c fsck.c)
list(TRANSFORM ARTB_FSCK_F2FS_SOURCES PREPEND "${ARTB_F2FS_ROOT}/fsck/")
add_executable(artb_sload_f2fs ${ARTB_FSCK_F2FS_SOURCES} "${ARTB_F2FS_ROOT}/fsck/sload.c")
target_compile_definitions(artb_sload_f2fs PRIVATE WITH_SLOAD)
# cutils goes first so its portable private/fs_config.h wins over the Android tree copy that needs linux/capability.h.
target_link_libraries(artb_sload_f2fs PRIVATE cutils_static artb_f2fs)
add_executable(artb_fsck_f2fs ${ARTB_FSCK_F2FS_SOURCES} "${ARTB_F2FS_ROOT}/fsck/defrag.c" "${ARTB_F2FS_ROOT}/fsck/resize.c")
target_compile_definitions(artb_fsck_f2fs PRIVATE WITH_DUMP WITH_DEFRAG WITH_RESIZE)
target_link_libraries(artb_fsck_f2fs PRIVATE artb_f2fs)
foreach(ARTB_F2FS_TOOL mkfs sload fsck)
    set_target_properties(artb_${ARTB_F2FS_TOOL}_f2fs PROPERTIES OUTPUT_NAME ${ARTB_F2FS_TOOL}.f2fs LINKER_LANGUAGE CXX)
    if(WIN32)
        target_sources(artb_${ARTB_F2FS_TOOL}_f2fs PRIVATE "${ARTB_UTF8_MANIFEST_RC}")
    endif()
endforeach()
find_package(Threads REQUIRED)
add_executable(extract.f2fs extract_f2fs.c)
target_link_libraries(extract.f2fs PRIVATE artb_cpu artb_extract_io lz4_static libzstd_static Threads::Threads)
if(WIN32)
    target_link_options(extract.f2fs PRIVATE -municode)
endif()
install(TARGETS extract.f2fs artb_mkfs_f2fs artb_sload_f2fs RUNTIME DESTINATION bin)
