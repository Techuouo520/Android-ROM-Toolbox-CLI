function(artb_mb_library name)
    add_library(${name} STATIC ${ARGN})
    target_compile_definitions(${name} PRIVATE ${MB_DEFINES})
    target_include_directories(${name} PRIVATE ${MB_INCLUDES})
    target_compile_options(${name} PRIVATE ${MB_OPTIONS})
endfunction()
