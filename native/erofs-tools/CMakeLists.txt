set(ARTB_EROFS_ROOT "${CMAKE_CURRENT_LIST_DIR}/../../upstream/erofs-tools")
set(ARTB_EROFS_UTILS_ROOT "${ARTB_EROFS_ROOT}/src/lib/erofs-utils")
if(NOT EXISTS "${ARTB_EROFS_ROOT}/src/extract/CMakeLists.txt" OR NOT EXISTS "${ARTB_EROFS_UTILS_ROOT}/lib/io.c")
    message(FATAL_ERROR "EROFS sources missing under ${ARTB_EROFS_ROOT}")
endif()
set(ARTB_EROFS_COMPAT_DIR "${CMAKE_CURRENT_LIST_DIR}/compat")
set(USE_LIBCXX OFF CACHE BOOL "" FORCE)
set(USE_LIBZ_STABLE_VERSION ON CACHE BOOL "" FORCE)
set(ENABLE_FULL_LTO OFF CACHE BOOL "" FORCE)
set(ENABLE_LIBFUSE OFF CACHE BOOL "" FORCE)
# Upstream raises EROFS_MAX_BLOCK_SIZE only for the aarch64 C library, so C++ callers allocated smaller erofs_buf structs than the library wrote; every target here must share one value.
set(MAX_BLOCK_SIZE 16384)
add_compile_definitions(EROFS_MAX_BLOCK_SIZE=${MAX_BLOCK_SIZE})
add_subdirectory("${ARTB_EROFS_ROOT}" "${CMAKE_CURRENT_BINARY_DIR}/upstream" EXCLUDE_FROM_ALL)
if(CMAKE_C_COMPILER_ID STREQUAL "GNU")
    target_compile_options(pcre2_static PRIVATE -Wno-error=maybe-uninitialized)
endif()
target_link_libraries(erofs_tool_static PUBLIC artb_cpu)
target_link_libraries(erofs_utils_static PUBLIC artb_cpu)
if(WIN32)
    target_sources(extract.erofs PRIVATE "${ARTB_UTF8_MANIFEST_RC}")
endif()
find_package(Threads REQUIRED)
set(ARTB_EROFS_CONFIG_DIR "${CMAKE_CURRENT_BINARY_DIR}/upstream/src/lib/liberofs")
file(STRINGS "${ARTB_EROFS_UTILS_ROOT}/VERSION" ARTB_EROFS_UTILS_VERSION LIMIT_COUNT 1)
set(ARTB_MKFS_EROFS_SOURCES config.c io.c cache.c super.c inode.c xattr.c exclude.c namei.c data.c compress.c compressor.c zmap.c decompress.c compress_hints.c hashmap.c sha256.c blobchunk.c dir.c fragments.c dedupe.c uuid_unparse.c uuid.c tar.c block_list.c rebuild.c diskbuf.c bitops.c dedupe_ext.c vmdk.c metabox.c global.c importer.c base64.c compressor_lz4.c compressor_lz4hc.c compressor_liblzma.c kite_deflate.c compressor_deflate.c compressor_libzstd.c gzran.c workqueue.c)
list(TRANSFORM ARTB_MKFS_EROFS_SOURCES PREPEND "${ARTB_EROFS_UTILS_ROOT}/lib/")
add_executable(artb_mkfs_erofs ${ARTB_MKFS_EROFS_SOURCES} "${ARTB_EROFS_UTILS_ROOT}/mkfs/main.c")
target_include_directories(artb_mkfs_erofs PRIVATE "${ARTB_EROFS_UTILS_ROOT}/include" "${ARTB_EROFS_UTILS_ROOT}/lib" "${ARTB_EROFS_CONFIG_DIR}" "${ARTB_EROFS_ROOT}/src/lib/xz/src/liblzma/api")
# SELinux labels come from the exact-path selabel shim; artb resolves file_contexts expressions before building.
target_compile_definitions(artb_mkfs_erofs PRIVATE _FILE_OFFSET_BITS=64 _LARGEFILE_SOURCE _LARGEFILE64_SOURCE HAVE_LIBSELINUX=1 "PACKAGE_VERSION=\"${ARTB_EROFS_UTILS_VERSION}\"")
target_compile_options(artb_mkfs_erofs PRIVATE "-include${ARTB_EROFS_CONFIG_DIR}/config.h" -Wno-ignored-qualifiers -Wno-pointer-arith -Wno-unused-parameter -Wno-unused-function -Wno-unused-label -Wno-deprecated-declarations -Wno-c99-designator)
if(WIN32)
    set(ARTB_EROFS_BASE_LINK base_static)
endif()
target_link_libraries(artb_mkfs_erofs PRIVATE xxhash cutils_static ${ARTB_EROFS_BASE_LINK} lz4_static liblzma z_static libzstd_static artb_cpu artb_selabel Threads::Threads)
if(WIN32)
    target_sources(artb_mkfs_erofs PRIVATE "${ARTB_EROFS_COMPAT_DIR}/artb_mkfs_windows_compat.c" "${ARTB_UTF8_MANIFEST_RC}")
    target_include_directories(artb_mkfs_erofs BEFORE PRIVATE "${ARTB_EROFS_COMPAT_DIR}/mkfs" "${ARTB_EROFS_COMPAT_DIR}")
    target_compile_options(artb_mkfs_erofs PRIVATE "-include${ARTB_EROFS_COMPAT_DIR}/artb_mkfs_windows_compat.h" "-include${ARTB_EROFS_COMPAT_DIR}/artb_erofs_windows_compat.h")
    # MinGW has no POSIX regex, so the vendored PCRE2's POSIX layer backs --exclude-regex and compression hints; artb passes neither.
    add_library(artb_pcre2posix STATIC "${ARTB_EROFS_ROOT}/src/lib/pcre/src/pcre2posix.c")
    target_compile_definitions(artb_pcre2posix PUBLIC PCRE2_STATIC PRIVATE HAVE_CONFIG_H PCRE2_CODE_UNIT_WIDTH=8)
    target_include_directories(artb_pcre2posix PUBLIC "${ARTB_EROFS_COMPAT_DIR}/regex" PRIVATE "${ARTB_EROFS_ROOT}/src/lib/pcre/src")
    target_compile_definitions(pcre2_static PUBLIC PCRE2_STATIC)
    target_link_libraries(artb_pcre2posix PUBLIC pcre2_static)
    target_link_libraries(artb_mkfs_erofs PRIVATE artb_winfs artb_pcre2posix)
endif()
set_target_properties(artb_mkfs_erofs PROPERTIES OUTPUT_NAME mkfs.erofs LINKER_LANGUAGE CXX)
