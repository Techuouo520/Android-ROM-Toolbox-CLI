set(ARTB_E2FSPROGS_ROOT "${CMAKE_CURRENT_LIST_DIR}/../../upstream/e2fsprogs" CACHE PATH "AOSP e2fsprogs source root")
if(NOT EXISTS "${ARTB_E2FSPROGS_ROOT}/lib/ext2fs/openfs.c")
    message(FATAL_ERROR "e2fsprogs sources missing at ${ARTB_E2FSPROGS_ROOT}")
endif()
if(NOT TARGET artb_sparse)
    message(FATAL_ERROR "EXT4 tools require ARTB_BUILD_ANDROID_TOOLS=ON for libsparse")
endif()
find_package(Threads REQUIRED)
set(E2FS_LIB "${ARTB_E2FSPROGS_ROOT}/lib")
set(E2FS_INCLUDES "${E2FS_LIB}")
if(WIN32)
    list(APPEND E2FS_INCLUDES "${ARTB_E2FSPROGS_ROOT}/include/mingw")
endif()
set(E2FS_OPTIONS -Wno-pointer-arith -Wno-sign-compare -Wno-type-limits -Wno-unused-parameter)
set(E2FS_ET_SOURCES error_message.c et_name.c init_et.c com_err.c com_right.c)
list(TRANSFORM E2FS_ET_SOURCES PREPEND "${E2FS_LIB}/et/")
set(E2FS_UUID_SOURCES clear.c compare.c copy.c gen_uuid.c isnull.c pack.c parse.c unpack.c unparse.c uuid_time.c)
list(TRANSFORM E2FS_UUID_SOURCES PREPEND "${E2FS_LIB}/uuid/")
set(E2FS_E2P_SOURCES encoding.c errcode.c feature.c fgetflags.c fsetflags.c fgetproject.c fsetproject.c fgetversion.c fsetversion.c getflags.c getversion.c hashstr.c iod.c ljs.c ls.c mntopts.c parse_num.c pe.c pf.c ps.c setflags.c setversion.c uuid.c ostype.c percent.c)
list(TRANSFORM E2FS_E2P_SOURCES PREPEND "${E2FS_LIB}/e2p/")
set(E2FS_EXT2FS_SOURCES ext2_err.c alloc.c alloc_sb.c alloc_stats.c alloc_tables.c atexit.c badblocks.c bb_inode.c bitmaps.c bitops.c blkmap64_ba.c blkmap64_rb.c blknum.c block.c bmap.c check_desc.c crc16.c crc32c.c csum.c closefs.c dblist.c dblist_dir.c digest_encode.c dirblock.c dirhash.c dir_iterate.c dupfs.c expanddir.c ext_attr.c extent.c fallocate.c fileio.c finddev.c flushb.c freefs.c gen_bitmap.c gen_bitmap64.c get_num_dirs.c get_pathname.c getsize.c getsectsize.c hashmap.c i_block.c icount.c imager.c ind_block.c initialize.c inline.c inline_data.c inode.c io_manager.c ismounted.c link.c llseek.c lookup.c mmp.c mkdir.c mkjournal.c namei.c native.c newdir.c nls_utf8.c openfs.c progress.c punch.c qcow2.c rbtree.c read_bb.c read_bb_file.c res_gdt.c rw_bitmaps.c sha256.c sha512.c swapfs.c symlink.c undo_io.c sparse_io.c unlink.c valid_blk.c version.c test_io.c)
if(WIN32)
    list(APPEND E2FS_EXT2FS_SOURCES windows_io.c)
else()
    list(APPEND E2FS_EXT2FS_SOURCES unix_io.c)
endif()
list(TRANSFORM E2FS_EXT2FS_SOURCES PREPEND "${E2FS_LIB}/ext2fs/")
add_library(artb_ext2_com_err STATIC ${E2FS_ET_SOURCES})
add_library(artb_ext2_uuid STATIC ${E2FS_UUID_SOURCES})
add_library(artb_ext2fs STATIC ${E2FS_EXT2FS_SOURCES})
add_library(artb_ext2_e2p STATIC ${E2FS_E2P_SOURCES})
foreach(ARTB_E2FS_TARGET artb_ext2_com_err artb_ext2_uuid artb_ext2fs artb_ext2_e2p)
    target_include_directories(${ARTB_E2FS_TARGET} PUBLIC ${E2FS_INCLUDES})
    target_compile_options(${ARTB_E2FS_TARGET} PRIVATE ${E2FS_OPTIONS})
endforeach()
target_link_libraries(artb_ext2fs PUBLIC artb_ext2_com_err artb_ext2_uuid artb_sparse ZLIB::ZLIB)
target_link_libraries(artb_ext2_e2p PUBLIC artb_ext2fs)
add_executable(extract.ext4 extract_ext4.c)
target_link_libraries(extract.ext4 PRIVATE artb_ext2_e2p artb_ext2fs artb_cpu artb_extract_io Threads::Threads)
set_target_properties(extract.ext4 PROPERTIES LINKER_LANGUAGE CXX)
if(WIN32)
    target_link_options(extract.ext4 PRIVATE -municode)
endif()
install(TARGETS extract.ext4 RUNTIME DESTINATION bin)
if(NOT TARGET cutils_static OR NOT TARGET base_static)
    message(FATAL_ERROR "EXT4 packing tools require ARTB_BUILD_EROFS_TOOLS=ON for libcutils and libbase")
endif()
set(E2FS_BLKID_SOURCES cache.c dev.c devname.c devno.c getsize.c llseek.c probe.c read.c resolve.c save.c tag.c version.c)
list(TRANSFORM E2FS_BLKID_SOURCES PREPEND "${E2FS_LIB}/blkid/")
set(E2FS_QUOTA_SOURCES devname.c dict.c mkquota.c parse_qtype.c plausible.c profile.c profile_helpers.c prof_err.c quotaio.c quotaio_tree.c quotaio_v2.c)
list(TRANSFORM E2FS_QUOTA_SOURCES PREPEND "${E2FS_LIB}/support/")
add_library(artb_ext2_blkid STATIC ${E2FS_BLKID_SOURCES})
add_library(artb_ext2_quota STATIC ${E2FS_QUOTA_SOURCES})
add_library(artb_ext2_misc STATIC "${ARTB_E2FSPROGS_ROOT}/misc/create_inode.c")
foreach(ARTB_E2FS_TARGET artb_ext2_blkid artb_ext2_quota artb_ext2_misc)
    target_include_directories(${ARTB_E2FS_TARGET} PUBLIC ${E2FS_INCLUDES})
    target_compile_options(${ARTB_E2FS_TARGET} PRIVATE ${E2FS_OPTIONS})
endforeach()
target_include_directories(artb_ext2_misc PUBLIC "${ARTB_E2FSPROGS_ROOT}/misc")
target_link_libraries(artb_ext2_blkid PUBLIC artb_ext2_uuid)
target_link_libraries(artb_ext2_quota PUBLIC artb_ext2fs artb_ext2_blkid artb_ext2_com_err)
target_link_libraries(artb_ext2_misc PUBLIC artb_ext2_quota artb_ext2fs)
if(WIN32)
    target_link_libraries(artb_ext2_misc PUBLIC artb_winfs)
endif()
add_executable(mke2fs "${ARTB_E2FSPROGS_ROOT}/misc/mke2fs.c" "${ARTB_E2FSPROGS_ROOT}/misc/util.c" "${ARTB_E2FSPROGS_ROOT}/misc/mk_hugefiles.c" "${ARTB_E2FSPROGS_ROOT}/misc/default_profile.c")
target_link_libraries(mke2fs PRIVATE artb_ext2_misc artb_ext2_quota artb_ext2_blkid artb_ext2_e2p artb_ext2fs artb_ext2_uuid artb_ext2_com_err Threads::Threads)
target_compile_definitions(mke2fs PRIVATE ARTB_MKE2FS)
set(E2FSDROID_SOURCES e2fsdroid.c block_range.c fsmap.c block_list.c base_fs.c perms.c basefs_allocator.c)
list(TRANSFORM E2FSDROID_SOURCES PREPEND "${ARTB_E2FSPROGS_ROOT}/contrib/android/")
add_executable(e2fsdroid ${E2FSDROID_SOURCES})
# The real fs_config and SELinux paths replace the Windows stubs in perms.h; labels come from the exact-path selabel shim.
target_compile_definitions(e2fsdroid PRIVATE ARTB_E2FSDROID HAVE_SYS_STAT_H=1)
# The erofs-tools libcutils headers build on hosts without linux/capability.h, so they must win over the android-tools copy.
get_target_property(ARTB_CUTILS_INCLUDES cutils_static INTERFACE_INCLUDE_DIRECTORIES)
target_include_directories(e2fsdroid BEFORE PRIVATE ${ARTB_CUTILS_INCLUDES} "${E2FS_LIB}/ext2fs")
if(WIN32)
    target_sources(e2fsdroid PRIVATE compat/artb_e2fsdroid_compat.c)
    target_include_directories(e2fsdroid PRIVATE compat)
    target_compile_options(e2fsdroid PRIVATE "-include${CMAKE_CURRENT_LIST_DIR}/compat/artb_e2fsdroid_compat.h")
endif()
if(WIN32)
    set(ARTB_E2FS_BASE_LINK base_static)
endif()
target_link_libraries(e2fsdroid PRIVATE artb_ext2_misc artb_ext2fs artb_ext2_com_err cutils_static ${ARTB_E2FS_BASE_LINK} artb_selabel Threads::Threads)
foreach(ARTB_E2FS_TOOL mke2fs e2fsdroid)
    target_compile_options(${ARTB_E2FS_TOOL} PRIVATE ${E2FS_OPTIONS})
    set_target_properties(${ARTB_E2FS_TOOL} PROPERTIES LINKER_LANGUAGE CXX)
    if(WIN32)
        target_sources(${ARTB_E2FS_TOOL} PRIVATE "${ARTB_UTF8_MANIFEST_RC}")
    endif()
endforeach()
install(TARGETS mke2fs e2fsdroid RUNTIME DESTINATION bin)
