cmake_minimum_required(VERSION 3.24)
set(ARTB_ANDROID_TOOLS_ROOT "${CMAKE_CURRENT_LIST_DIR}/../../upstream/android-tools/vendor" CACHE PATH "nmeum/android-tools vendor source root")
if(NOT EXISTS "${ARTB_ANDROID_TOOLS_ROOT}/core/libsparse/sparse.cpp")
    message(FATAL_ERROR "Android tools sources missing at ${ARTB_ANDROID_TOOLS_ROOT}")
endif()
if(ANDROID)
    # These are host tools on every OS: on a phone they must log to stderr and treat paths as image files, not logd, properties, init control files or /dev/block/by-name; the NDK's newer-API marks on liblog's own statically linked functions are relaxed.
    add_compile_options(-U__ANDROID__ -Wno-unguarded-availability)
    add_compile_definitions(__ANDROID_UNAVAILABLE_SYMBOLS_ARE_WEAK__)
endif()
add_subdirectory("${ARTB_ANDROID_TOOLS_ROOT}/fmtlib" "${CMAKE_CURRENT_BINARY_DIR}/fmt" EXCLUDE_FROM_ALL)
add_compile_definitions(_FILE_OFFSET_BITS=64 _LARGEFILE64_SOURCE)
if(WIN32)
    add_compile_options("-include${CMAKE_CURRENT_LIST_DIR}/compat/artb_windows_compat.h")
endif()
set(ANDROID_INCLUDE_DIRS "${ARTB_ANDROID_TOOLS_ROOT}/core/include" "${ARTB_ANDROID_TOOLS_ROOT}/core/libcutils/include" "${ARTB_ANDROID_TOOLS_ROOT}/libbase/include" "${ARTB_ANDROID_TOOLS_ROOT}/logging/liblog/include" "${CMAKE_CURRENT_LIST_DIR}/compat")
add_library(artb_android_log STATIC "${ARTB_ANDROID_TOOLS_ROOT}/logging/liblog/log_event_list.cpp" "${ARTB_ANDROID_TOOLS_ROOT}/logging/liblog/log_event_write.cpp" "${ARTB_ANDROID_TOOLS_ROOT}/logging/liblog/logger_name.cpp" "${ARTB_ANDROID_TOOLS_ROOT}/logging/liblog/logger_read.cpp" "${ARTB_ANDROID_TOOLS_ROOT}/logging/liblog/logger_write.cpp" "${ARTB_ANDROID_TOOLS_ROOT}/logging/liblog/properties.cpp" "${ARTB_ANDROID_TOOLS_ROOT}/logging/liblog/logprint.cpp")
target_compile_definitions(artb_android_log PRIVATE LIBLOG_LOG_TAG=1006 FAKE_LOG_DEVICE=1)
if(NOT APPLE)
    target_compile_definitions(artb_android_log PRIVATE _XOPEN_SOURCE=700)
endif()
target_include_directories(artb_android_log PUBLIC ${ANDROID_INCLUDE_DIRS})
set(ANDROID_BASE_SOURCES "${ARTB_ANDROID_TOOLS_ROOT}/libbase/chrono_utils.cpp" "${ARTB_ANDROID_TOOLS_ROOT}/libbase/file.cpp" "${ARTB_ANDROID_TOOLS_ROOT}/libbase/hex.cpp" "${ARTB_ANDROID_TOOLS_ROOT}/libbase/logging.cpp" "${ARTB_ANDROID_TOOLS_ROOT}/libbase/mapped_file.cpp" "${ARTB_ANDROID_TOOLS_ROOT}/libbase/parsebool.cpp" "${ARTB_ANDROID_TOOLS_ROOT}/libbase/parsenetaddress.cpp" "${ARTB_ANDROID_TOOLS_ROOT}/libbase/posix_strerror_r.cpp" "${ARTB_ANDROID_TOOLS_ROOT}/libbase/process.cpp" "${ARTB_ANDROID_TOOLS_ROOT}/libbase/properties.cpp" "${ARTB_ANDROID_TOOLS_ROOT}/libbase/result.cpp" "${ARTB_ANDROID_TOOLS_ROOT}/libbase/stringprintf.cpp" "${ARTB_ANDROID_TOOLS_ROOT}/libbase/strings.cpp" "${ARTB_ANDROID_TOOLS_ROOT}/libbase/threads.cpp")
if(WIN32)
    list(APPEND ANDROID_BASE_SOURCES "${ARTB_ANDROID_TOOLS_ROOT}/libbase/errors_windows.cpp" "${ARTB_ANDROID_TOOLS_ROOT}/libbase/utf8.cpp")
else()
    list(APPEND ANDROID_BASE_SOURCES "${ARTB_ANDROID_TOOLS_ROOT}/libbase/errors_unix.cpp")
endif()
add_library(artb_android_base STATIC ${ANDROID_BASE_SOURCES})
target_include_directories(artb_android_base PUBLIC ${ANDROID_INCLUDE_DIRS})
target_link_libraries(artb_android_base PUBLIC fmt::fmt artb_android_log)
add_library(artb_sparse STATIC "${ARTB_ANDROID_TOOLS_ROOT}/core/libsparse/backed_block.cpp" "${ARTB_ANDROID_TOOLS_ROOT}/core/libsparse/output_file.cpp" "${ARTB_ANDROID_TOOLS_ROOT}/core/libsparse/sparse.cpp" "${ARTB_ANDROID_TOOLS_ROOT}/core/libsparse/sparse_crc32.cpp" "${ARTB_ANDROID_TOOLS_ROOT}/core/libsparse/sparse_err.cpp" "${ARTB_ANDROID_TOOLS_ROOT}/core/libsparse/sparse_read.cpp")
target_include_directories(artb_sparse PUBLIC "${ARTB_ANDROID_TOOLS_ROOT}/core/libsparse/include" ${ANDROID_INCLUDE_DIRS})
# Headers only: linking zlib here would put the system copy ahead of the vendored one in the F2FS tools' link lines.
target_include_directories(artb_sparse PRIVATE ${ZLIB_INCLUDE_DIRS})
target_link_libraries(artb_sparse PRIVATE artb_android_base)
add_executable(simg2img "${ARTB_ANDROID_TOOLS_ROOT}/core/libsparse/simg2img.cpp")
target_link_libraries(simg2img PRIVATE artb_sparse ZLIB::ZLIB artb_android_base)
add_executable(img2simg "${ARTB_ANDROID_TOOLS_ROOT}/core/libsparse/img2simg.cpp")
target_link_libraries(img2simg PRIVATE artb_sparse ZLIB::ZLIB artb_android_base)
add_library(artb_ext4 STATIC "${ARTB_ANDROID_TOOLS_ROOT}/extras/ext4_utils/ext4_utils.cpp" "${ARTB_ANDROID_TOOLS_ROOT}/extras/ext4_utils/wipe.cpp" "${ARTB_ANDROID_TOOLS_ROOT}/extras/ext4_utils/ext4_sb.cpp")
target_include_directories(artb_ext4 PUBLIC "${ARTB_ANDROID_TOOLS_ROOT}/extras/ext4_utils/include" ${ANDROID_INCLUDE_DIRS})
target_link_libraries(artb_ext4 PRIVATE artb_android_base artb_sparse)
set(LIBLP_SOURCES "${ARTB_ANDROID_TOOLS_ROOT}/fs_mgr/liblp/builder.cpp" "${ARTB_ANDROID_TOOLS_ROOT}/fs_mgr/liblp/super_layout_builder.cpp" "${ARTB_ANDROID_TOOLS_ROOT}/fs_mgr/liblp/images.cpp" "${ARTB_ANDROID_TOOLS_ROOT}/fs_mgr/liblp/partition_opener.cpp" "${ARTB_ANDROID_TOOLS_ROOT}/fs_mgr/liblp/property_fetcher.cpp" "${ARTB_ANDROID_TOOLS_ROOT}/fs_mgr/liblp/reader.cpp" "${ARTB_ANDROID_TOOLS_ROOT}/fs_mgr/liblp/utility.cpp" "${ARTB_ANDROID_TOOLS_ROOT}/fs_mgr/liblp/writer.cpp")
add_library(artb_lp STATIC ${LIBLP_SOURCES})
target_include_directories(artb_lp PUBLIC "${ARTB_ANDROID_TOOLS_ROOT}/fs_mgr/liblp/include" "${ARTB_ANDROID_TOOLS_ROOT}/extras/ext4_utils/include" ${ANDROID_INCLUDE_DIRS})
target_link_libraries(artb_lp PRIVATE artb_android_base artb_sparse artb_ext4 artb_sha256 ZLIB::ZLIB)
add_executable(lpunpack "${ARTB_ANDROID_TOOLS_ROOT}/extras/partition_tools/lpunpack.cc")
target_include_directories(lpunpack PRIVATE ${ANDROID_INCLUDE_DIRS})
target_link_libraries(lpunpack PRIVATE artb_android_base artb_android_log artb_lp artb_sparse artb_cpu artb_sparse_reader)
add_executable(lpmake "${ARTB_ANDROID_TOOLS_ROOT}/extras/partition_tools/lpmake.cc")
target_include_directories(lpmake PRIVATE ${ANDROID_INCLUDE_DIRS})
target_link_libraries(lpmake PRIVATE artb_android_base artb_android_log artb_lp artb_sparse)
if(WIN32)
    # Binary mode and the UTF-8 code page keep image data intact and let std::filesystem accept Unicode paths.
    foreach(ARTB_NATIVE_TARGET simg2img img2simg lpunpack lpmake)
        target_sources(${ARTB_NATIVE_TARGET} PRIVATE "${CMAKE_CURRENT_LIST_DIR}/compat/artb_binary_mode.cpp" "${ARTB_UTF8_MANIFEST_RC}")
    endforeach()
endif()
install(TARGETS simg2img img2simg lpunpack lpmake RUNTIME DESTINATION bin)
